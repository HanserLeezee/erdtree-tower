cmake_minimum_required(VERSION 3.20)
project(TowerSeamlessGate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SDK "${STEAMWORKS_ROOT}/include/steam")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_forwarders.py" "${SDK}" "${CMAKE_CURRENT_BINARY_DIR}/forwarders.h" COMMAND_ERROR_IS_FATAL ANY)
set(MINHOOK "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/hudhook/vendor/minhook")
add_library(TowerSeamlessGate SHARED gate.cpp interceptor.cpp version.rc "${MINHOOK}/src/hook.c" "${MINHOOK}/src/buffer.c" "${MINHOOK}/src/trampoline.c" "${MINHOOK}/src/hde/hde64.c")
set_target_properties(TowerSeamlessGate PROPERTIES OUTPUT_NAME "Tower.SeamlessGate")
target_include_directories(TowerSeamlessGate PRIVATE "${SDK}" "${CMAKE_CURRENT_BINARY_DIR}" "${MINHOOK}/include")
target_compile_definitions(TowerSeamlessGate PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(TowerSeamlessGate PRIVATE /utf-8 /wd4828)
target_link_libraries(TowerSeamlessGate PRIVATE bcrypt)
add_executable(gate-tests tests.cpp)
target_include_directories(gate-tests PRIVATE "${SDK}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(gate-tests PRIVATE /utf-8 /wd4828 /UNDEBUG)
target_compile_definitions(gate-tests PRIVATE _CRT_SECURE_NO_WARNINGS)
enable_testing()
add_test(NAME admission COMMAND gate-tests)

add_library(gate-test-steam SHARED hook_fixture.cpp)
set_target_properties(gate-test-steam PROPERTIES OUTPUT_NAME steam_api64)
add_library(gate-test-ersc SHARED ersc_fixture.cpp)
set_target_properties(gate-test-ersc PROPERTIES OUTPUT_NAME ersc)
target_link_libraries(gate-test-ersc PRIVATE gate-test-steam)
add_executable(gate-hook-tests hook_test.cpp)
foreach(target gate-test-steam gate-test-ersc gate-hook-tests)
    target_include_directories(${target} PRIVATE "${SDK}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(${target} PRIVATE /utf-8 /wd4828 /UNDEBUG)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/abi/$<CONFIG>")
endforeach()
add_dependencies(gate-hook-tests gate-test-ersc TowerSeamlessGate)
add_custom_command(TARGET gate-hook-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:TowerSeamlessGate>" "$<TARGET_FILE_DIR:gate-hook-tests>/Tower.SeamlessGate.dll")
add_test(NAME factory-loader-lock COMMAND gate-hook-tests)
set_tests_properties(factory-loader-lock PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:gate-hook-tests>" TIMEOUT 20)
